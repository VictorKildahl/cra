// Sources/LocalStorage/TabDataModel/TabDataModelSchemaV14.swift from github.com/phibrowser/phibrowser-mac
// (8a4fcd2, 19 Nov 2026), so the tables are the ones Phi writes:
//   swiftc -o phigen TabDataModelSchemaV14.swift make-store.swift && ./phigen <out>/LocalStore.sqlite space|profile|app
// then: sqlite3 <file> "PRAGMA wal_checkpoint(TRUNCATE); PRAGMA journal_mode=DELETE; VACUUM;"

import Foundation
import SwiftData

typealias TabDataModel = TabDataModelSchemaV14.TabDataModel
typealias ProfileModel = TabDataModelSchemaV14.ProfileModel
typealias SpaceModel = TabDataModelSchemaV14.SpaceModel
typealias BrowserDataSettingsModel = TabDataModelSchemaV14.BrowserDataSettingsModel

let out = URL(fileURLWithPath: CommandLine.arguments[1])
let scope = CommandLine.arguments.count > 2 ? CommandLine.arguments[2] : "space"
try? FileManager.default.createDirectory(at: out.deletingLastPathComponent(), withIntermediateDirectories: true)
let schema = Schema(versionedSchema: TabDataModelSchemaV14.self)
let config = ModelConfiguration(schema: schema, url: out)
let container = try ModelContainer(for: schema, configurations: [config])
let ctx = ModelContext(container)
let t0 = Date(timeIntervalSince1970: 1_790_000_000)

func row(_ title: String, _ url: String, _ type: Int, _ index: Int, space: String? = nil, profile: String? = nil) -> TabDataModel {
    let r = TabDataModel(title: title, guid: UUID().uuidString, index: index, url: URL(string: url)!, favicon: nil, createdDate: t0, updatedDate: t0)
    r.type = type; r.spaceId = space; r.profileId = profile
    ctx.insert(r); return r
}
func root(_ space: String, _ profile: String) -> TabDataModel { row("", "phi://bookmarks-root", 3, 0, space: space, profile: profile) }
